import Foundation

/// Descriptor unificado para focus types (built-in o custom)
enum FocusTypeDescriptor: Equatable, Codable {
    case builtIn(FocusModeType)
    case custom(FocusType)

    var id: UUID {
        switch self {
        case .builtIn(let type):
            return UUID(uuidString: type.rawValue) ?? UUID()
        case .custom(let type):
            return type.id
        }
    }

    var name: String {
        switch self {
        case .builtIn(let type):
            return AppLanguage.localizedString(type.localizedLabel)
        case .custom(let type):
            return type.name
        }
    }

    var emoji: String {
        switch self {
        case .builtIn:
            return ""  // Built-in types don't have emojis
        case .custom(let type):
            return type.emoji
        }
    }

    var modeType: FocusModeType {
        switch self {
        case .builtIn(let type):
            return type
        case .custom:
            return .userCustom
        }
    }
}

import Foundation
import Observation

@MainActor
@Observable
final class FocusTypesService {
    static let shared = FocusTypesService()

    private static let customTypesKey = "focally.focusTypes.custom"
    private(set) var customTypes: [FocusType] = []

    private init() {
        loadCustomTypes()
    }

    // MARK: - CRUD

    func addCustomType(_ type: FocusType) {
        customTypes.append(type)
        saveCustomTypes()
    }

    func updateCustomType(_ type: FocusType) {
        if let index = customTypes.firstIndex(where: { $0.id == type.id }) {
            customTypes[index] = type
            saveCustomTypes()
        }
    }

    func deleteCustomType(id: UUID) {
        customTypes.removeAll { $0.id == id }
        saveCustomTypes()
    }

    // MARK: - Query

    func getAllDescriptors() -> [FocusTypeDescriptor] {
        var descriptors: [FocusTypeDescriptor] = []

        // Built-in types
        for type in FocusModeType.allCases where type != .userCustom {
            descriptors.append(.builtIn(type))
        }

        // Custom types
        for type in customTypes {
            descriptors.append(.custom(type))
        }

        return descriptors
    }

    func findDescriptor(id: UUID) -> FocusTypeDescriptor? {
        // Check built-in types
        for type in FocusModeType.allCases {
            if UUID(uuidString: type.rawValue) == id {
                return .builtIn(type)
            }
        }

        // Check custom types
        for type in customTypes {
            if type.id == id {
                return .custom(type)
            }
        }

        return nil
    }

    // MARK: - Persistence

    private func loadCustomTypes() {
        guard let data = UserDefaults.standard.data(forKey: Self.customTypesKey) else {
            return
        }
        customTypes = (try? JSONDecoder().decode([FocusType].self, from: data)) ?? []
    }

    private func saveCustomTypes() {
        guard let data = try? JSONEncoder().encode(customTypes) else { return }
        UserDefaults.standard.set(data, forKey: Self.customTypesKey)
    }
}

import Foundation
import EventKit

@MainActor
final class CalendarEventAnalyzer {
    static let shared = CalendarEventAnalyzer()

    private init() {}

    /// Detecta si un evento es una videollamada
    func isVideoCall(_ event: EKEvent) -> Bool {
        // Check 1: Location contiene URL de video call
        if let location = event.location?.lowercased() {
            let videoCallKeywords = [
                "zoom.us", "meet.google.com", "teams.microsoft.com",
                "webex.com", "gotomeeting.com", "bluejeans.com",
                "zoom", "meet", "teams", "webex"
            ]
            for keyword in videoCallKeywords {
                if location.contains(keyword) {
                    return true
                }
            }
        }

        // Check 2: URL property
        if let url = event.url?.absoluteString.lowercased() {
            let videoCallKeywords = [
                "zoom.us", "meet.google.com", "teams.microsoft.com",
                "webex.com"
            ]
            for keyword in videoCallKeywords {
                if url.contains(keyword) {
                    return true
                }
            }
        }

        // Check 3: Keywords en título
        if let title = event.title?.lowercased() {
            let titleKeywords = ["meet", "zoom", "call", "video", "teams"]
            for keyword in titleKeywords {
                if title.contains(keyword) {
                    return true
                }
            }
        }

        return false
    }
}

import Foundation

@MainActor
final class EmojiExtractor {
    static let shared = EmojiExtractor()

    private init() {}

    /// Extrae todos los emojis Unicode de un string
    func extractAllEmojis(from text: String) -> [String] {
        var emojis: [String] = []
        var currentScalar: UnicodeScalar?

        for scalar in text.unicodeScalars {
            if scalar.properties.isEmoji {
                // Iniciar o continuar emoji
                if currentScalar == nil {
                    currentScalar = scalar
                }
            } else {
                // Si terminó un emoji, agrégalo
                if let emoji = currentScalar {
                    emojis.append(String(emoji))
                    currentScalar = nil
                }
            }
        }

        // No agregar el último si sigue siendo un emoji (incompleto)
        return emojis
    }

    /// Concatena un array de emojis en un solo string
    func concatenate(emojis: [String]) -> String {
        return emojis.joined()
    }
}

private func determineChannel() -> SlackChannel {

    }

    // Priority 2: Calendar event
    if let activeEvent = activeCalendarEvent,

// MODIFICAR: buildSlackStatus()
func buildSlackStatus() -> SlackStatus {
    switch determineChannel() {
    case .idle:
        return buildIdleStatus()
    case .focusMode(let mode):
        return buildFocusModeStatus(mode)
    case .calendar(let event):
        return buildCalendarStatus(event)
    }

// NUEVO: buildCalendarStatus()
private func buildCalendarStatus(_ event: EKEvent) -> SlackStatus {
    let settings = slackSettings.calendarSettings

    // Emoji
    var emoji: String?
    if settings.useEventEmojisForStatus {
        let emojis = EmojiExtractor.shared.extractAllEmojis(from: event.title ?? "")
        if !emojis.isEmpty {
            emoji = EmojiExtractor.shared.concatenate(emojis: emojis)
        }
    } else {
        emoji = nil  // No default emoji for calendar channel
    }

    // Text
    var text: String?
    switch settings.titleDisplay {
    case .showFullTitle:
        text = event.title
    case .showVideoCallOnly:
        if CalendarEventAnalyzer.shared.isVideoCall(event) {
            text = AppLanguage.localizedString("calendar_in_video_call")
        } else {
            text = AppLanguage.localizedString("calendar_in_meeting")
        }
    case .hideTitle:
        text = nil
    }

    // DND
    var activateDND = false
    if settings.activateDNDForVideoCalls {
        activateDND = CalendarEventAnalyzer.shared.isVideoCall(event)
    }

    // Metrics tracking (solo para video calls del calendario)
    if CalendarEventAnalyzer.shared.isVideoCall(event) {
        let calendarVideoCallRecord = FocusSessionRecord(
            modeType: .calendarVideoCall,
            modeID: UUID(),
            startTime: event.startDate,
            endTime: event.endDate,
            duration: event.duration,
            pomodorosCompleted: nil
        )
        FocusMetricsService.shared.recordSession(calendarVideoCallRecord)
    }

    return SlackStatus(emoji: emoji, text: text, activateDND: activateDND)
}
